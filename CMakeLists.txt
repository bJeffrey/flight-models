cmake_minimum_required(VERSION 3.16)
project(flight_models LANGUAGES CXX)

# Configuration is reusable without any transport, dynamics or process runtime.
find_package(nlohmann_json REQUIRED)
find_path(VALIJSON_INCLUDE_DIR valijson/schema.hpp REQUIRED)
add_library(flight_models_config src/motion_profile.cpp)
target_compile_features(flight_models_config PUBLIC cxx_std_17)
target_compile_options(flight_models_config PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(flight_models_config PUBLIC include PRIVATE ${VALIJSON_INCLUDE_DIR})
target_link_libraries(flight_models_config PRIVATE nlohmann_json::nlohmann_json)

# Reuse the shared row-major frame types rather than introducing backend types.
set(FLIGHT_MATH_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../math" CACHE PATH "Shared math repository")
set(MATH_BUILD_TESTS OFF CACHE BOOL "Build shared math's independent tests")
add_subdirectory("${FLIGHT_MATH_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/shared-math" EXCLUDE_FROM_ALL)
add_library(flight_models_state src/state_store.cpp)
target_compile_features(flight_models_state PUBLIC cxx_std_17)
target_compile_options(flight_models_state PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(flight_models_state PUBLIC include)
target_link_libraries(flight_models_state PUBLIC math::math)

# The first scalar reference kernel keeps all numerical-backend details private.
add_library(flight_models_motion src/bounded_motion.cpp src/motion_driver.cpp)
target_compile_features(flight_models_motion PUBLIC cxx_std_17)
target_compile_options(flight_models_motion PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(flight_models_motion PUBLIC include)
target_link_libraries(flight_models_motion PUBLIC flight_models_state flight_models_config)

# Upstream's package config creates aliases for both DIS versions unconditionally.
# Import its exported targets directly so the image can install only DIS7.
find_package(OpenDIS CONFIG REQUIRED CONFIGS OpenDISTargets.cmake)
add_library(flight_models_dis src/dis_entity_state.cpp)
target_compile_features(flight_models_dis PUBLIC cxx_std_17)
target_compile_options(flight_models_dis PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(flight_models_dis PUBLIC include)
target_link_libraries(flight_models_dis PUBLIC flight_models_state PRIVATE OpenDIS::OpenDIS7)

# Reported navigation is a separate state owner; the PRD adapter never reads truth.
add_library(flight_models_navigation src/navigation.cpp src/position_report.cpp)
target_compile_features(flight_models_navigation PUBLIC cxx_std_17)
target_compile_options(flight_models_navigation PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(flight_models_navigation PUBLIC include)
target_link_libraries(flight_models_navigation PUBLIC flight_models_state flight_models_config nlohmann_json::nlohmann_json)

# Runtime orchestration composes domain libraries without adding transports to them.
add_library(flight_models_runtime src/runtime.cpp)
target_compile_features(flight_models_runtime PUBLIC cxx_std_17)
target_compile_options(flight_models_runtime PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(flight_models_runtime PUBLIC include)
target_link_libraries(flight_models_runtime PUBLIC flight_models_motion flight_models_dis flight_models_navigation)

# Domain libraries remain free of protobuf, NATS, WebSocket and process lifecycle.
set(_runtime_support_default "${CMAKE_CURRENT_SOURCE_DIR}/../ai-bm-sim/libraries/runtime-support")
if(DEFINED ENV{RUNTIME_SUPPORT_SOURCE_DIR})
  set(_runtime_support_default "$ENV{RUNTIME_SUPPORT_SOURCE_DIR}")
endif()
set(RUNTIME_SUPPORT_SOURCE_DIR "${_runtime_support_default}"
    CACHE PATH "Shared runtime-support source directory")
set(RUNTIME_SUPPORT_WITH_SIM_TIME ON CACHE BOOL "Build the shared simulation-time adapter")
add_subdirectory("${RUNTIME_SUPPORT_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/runtime-support" EXCLUDE_FROM_ALL)
find_package(Boost REQUIRED COMPONENTS system)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NATS REQUIRED IMPORTED_TARGET libnats)
add_library(flight_models_process src/process_config.cpp src/runtime_transport.cpp)
target_compile_features(flight_models_process PUBLIC cxx_std_17)
target_compile_options(flight_models_process PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(flight_models_process PUBLIC include PRIVATE ${VALIJSON_INCLUDE_DIR})
target_link_libraries(flight_models_process PUBLIC flight_models_runtime RuntimeSupport::sim_time
    Boost::system Threads::Threads PkgConfig::NATS)
add_executable(flight-models-run src/runtime_main.cpp)
target_compile_options(flight-models-run PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(flight-models-run PRIVATE flight_models_process)
